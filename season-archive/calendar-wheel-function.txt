function calendarGraph(g){
 const total=g.months.reduce((n,r)=>n+r.count,0),max=Math.max(1,...g.months.map(r=>r.count));
 const palette=['#dc7974','#e8a26a','#e6c56f','#9abf75','#76a5af','#7fa8dd','#b4a7d6','#d292b8','#80bda8','#b3b886','#bb9c80','#90a8bb'];
 const tip=(r,label)=>chartTip(label+'\n'+r.count+' seasons · '+(total?100*r.count/total:0).toFixed(1)+'%'+(r.seasons.length?'\nSeasons: '+r.seasons.map(s=>'S'+s).join(', '):''));
 const point=(angle,r)=>[180+Math.sin(angle)*r,180-Math.cos(angle)*r];
 const sector=(a,b,inner,outer)=>{const p=point(a,outer),q=point(b,outer),s=point(b,inner),t=point(a,inner);return `M${p} A${outer} ${outer} 0 0 1 ${q} L${s} A${inner} ${inner} 0 0 0 ${t} Z`;};
 const monthArcs=g.months.map((r,i)=>{const a=i*Math.PI/6,b=a+.215,c=a-.215,outer=78+70*r.count/max,label=point(a,167),value=point(a,outer+10);return `<g class="calendar-month"><path d="${sector(c,b,78,148)}" fill="#ffffff06"/><path class="calendar-wheel-segment" d="${sector(c,b,76,Math.max(79,outer))}" fill="${palette[i]}" ${tip(r,r.month)}/><text class="calendar-wheel-label" x="${label[0]}" y="${label[1]}" text-anchor="middle" dominant-baseline="middle">${r.month.slice(0,3)}</text><text class="calendar-wheel-value" x="${value[0]}" y="${value[1]}" text-anchor="middle" dominant-baseline="middle">${r.count}</text></g>`;}).join('');
 const center=`<text class="team-size-total" x="180" y="180" text-anchor="middle">${total}</text><text class="team-size-center-label" x="180" y="202" text-anchor="middle">SEASONS</text>`;
 const legend=(rows,key)=>`<div class="calendar-wheel-key">${rows.map((r,i)=>`<div ${tip(r,r[key])}><i style="background:${palette[i]}"></i><span>${escape(r[key])}</span><b>${r.count}</b><small>${(total?100*r.count/total:0).toFixed(1)}%</small></div>`).join('')}</div>`;
 const weekSvg=teamSizeGraph(g).match(/<svg[\s\S]*?<\/svg>/)[0];
 return `<div class="calendar-wheels"><section class="calendar-wheel-card"><div class="calendar-wheel-heading"><span>DAY OF WEEK DISTRIBUTION</span><h3>The week</h3><p>Seven days. When do we play?</p></div>${weekSvg}${legend(g.data,'day')}<p class="chart-caption">Segment size shows each day's share of seasons.</p></section><section class="calendar-wheel-card"><div class="calendar-wheel-heading"><span>MONTH OF YEAR DISTRIBUTION</span><h3>The year</h3><p>Twelve months. Our season for seasons.</p></div><svg class="calendar-year-wheel" viewBox="0 0 360 360" role="group" aria-label="Seasons by month, January at the top, clockwise through December">${[78,113,148].map(r=>`<circle cx="180" cy="180" r="${r}" fill="none" stroke="#ffffff0a" stroke-dasharray="2 4"/>`).join('')}${monthArcs}${center}</svg>${legend(g.months,'month')}<p class="chart-caption">Bar length shows season count. January starts at the top; read clockwise.</p></section></div><p class="chart-caption">Recorded match dates across all years, counted once per season in each wheel. Hover, focus, or tap a segment or legend entry for counts, shares, and season numbers.</p>`;
}

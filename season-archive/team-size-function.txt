function teamSizeGraph(g){
 const total=g.data.reduce((n,x)=>n+x.count,0),palette=['#dc7974','#e8a26a','#e6c56f','#9abf75','#76a5af','#7fa8dd','#b4a7d6'];
 let angle=-Math.PI/2;
 const arcs=g.data.map((r,i)=>{const start=angle;angle+=r.count/total*Math.PI*2;const end=angle,c=palette[i%palette.length],x=(a,radius)=>180+radius*Math.cos(a),y=(a,radius)=>180+radius*Math.sin(a),large=end-start>Math.PI?1:0;
  return `<path class="team-size-arc" d="M${x(start,142)} ${y(start,142)} A142 142 0 ${large} 1 ${x(end,142)} ${y(end,142)} L${x(end,100)} ${y(end,100)} A100 100 0 ${large} 0 ${x(start,100)} ${y(start,100)} Z" fill="${c}" stroke="#191c21" stroke-width="3" ${chartTip(`${r.size===1?'Solo':`Teams of ${r.size}`}\n${r.count} seasons · ${(r.count/total*100).toFixed(1)}%`)}/>`;
 }).join('');
 return `<div class="outcome-visual"><div class="outcome-chart-meta"><span>TEAM SIZE DISTRIBUTION</span><span>${total} SEASONS · ALL RECORDED MATCHES</span></div><div class="team-size-layout"><svg class="team-size-donut" viewBox="0 0 360 360" role="group" aria-label="Configured team sizes across ${total} seasons">${arcs}<text class="team-size-total" x="180" y="180" text-anchor="middle">${total}</text><text class="team-size-center-label" x="180" y="204" text-anchor="middle">SEASONS</text></svg><div class="team-size-breakdown"><div class="team-size-breakdown-heading"><span>Format</span><span>Seasons</span><span>Share</span></div>${g.data.map((r,i)=>`<div class="team-size-breakdown-row"><span class="team-size-format"><i style="background:${palette[i%palette.length]}"></i>${r.size===1?'Solo':`Teams of ${r.size}`}</span><b>${r.count}</b><span>${(100*r.count/total).toFixed(1)}%</span><div class="team-size-share-track" aria-hidden="true"><i style="width:${100*r.count/total}%;background:${palette[i%palette.length]}"></i></div></div>`).join('')}</div></div></div><p class="chart-caption">Distribution of configured team sizes, counted once per season. Hover, focus, or tap a segment for its count and share.</p>`;
}
